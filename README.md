### How I Work

```python
class AIEngineer:
    def __init__(self):
        self.principle = "if it isn't deployed, it isn't done"

    def build(self, idea):
        spec      = self.define_contracts(idea)     # types first
        prototype = self.experiment(spec)           # fast, ugly, honest
        metrics   = self.evaluate(prototype)        # no metric, no merge
        service   = self.refactor(prototype)        # modular, tested
        return self.ship(service)                   # docker → cloud → monitor

    def debug(self, model):
        while model.is_overfitting():
            self.question_the_data()   # it's almost always the data
        return model
```
